Appendix A — ЛИСТИНГ ПРОГРАММНОЙ РЕАЛИЗАЦИИ ПРОТОТИПА

В приложении А приведены основные программные модули прототипа многоуровневой системы защиты пользователей от вредоносной сетевой активности. Листинги отражают серверную часть, а именно модуль клиент-серверного взаимодействия, подсистема поведенческого профилирования и вспомогательные скрипты. Конфиденциальные значения токенов и IP-адреса скрыты или приведены неактуальные значения.

A.1 Cкрипт парсинга status-server.log

import sqlite3
import sys
import os
from datetime import datetime

DB_PATH = '/var/db/traffic.sqlite'
STATUS_LOG = '/run/openvpn-server/status-server.log'

def parse_status_file(filepath):
    users = {}
    try:
        with open(filepath, 'r') as f:
            for line in f:
                if line.startswith('CLIENT_LIST,'):
                    parts = line.strip().split(',')
                    if len(parts) >= 4:
                        common_name = parts[1]
                        virtual_addr = parts[3]
                        if virtual_addr and common_name and common_name != 'UNDEF':
                            users[common_name] = virtual_addr
    except Exception as e:
        print(f"Ошибка чтения файла статуса: {e}", file=sys.stderr)
        return None
    return users


def update_db(conn, current_users):
    cursor = conn.cursor()

    cursor.execute('SELECT username, ip_addr FROM vpn_users')
    db_records = {row[0]: row[1] for row in cursor.fetchall()}

    updated = 0

    for username, ip in current_users.items():
        if username in db_records:
            cursor.execute('''
                UPDATE vpn_users
                SET ip_addr = ?
                WHERE username = ?
            ''', (ip, username))
            updated += 1

    conn.commit()
    return updated


def main():
    db_dir = os.path.dirname(DB_PATH)
    if db_dir and not os.path.exists(db_dir):
        os.makedirs(db_dir, exist_ok=True)

    users = parse_status_file(STATUS_LOG)

    if users is None:
        print(f"{datetime.now().isoformat()}: Ошибка чтения файла статуса, обновление отменено", file=sys.stderr)
        sys.exit(1)

    conn = sqlite3.connect(DB_PATH)

    updated = update_db(conn, users)

    print(f"{datetime.now().isoformat()}: Обновлено: {updated}")
    conn.close()


if __name__ == '__main__':
    main()

import sqlite3
import time
from datetime import datetime
from constants import DB_PATH

def collector():
    ts_current = time.time()
    ts_back_6h = ts_current - 6 * 3600
    
    dt_window = datetime.fromtimestamp(ts_back_6h)
    weekday = dt_window.weekday()
    slot = dt_window.hour // 6

    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()

    cursor.execute('''SELECT vpn_users.id, vpn_users.ip_addr, vpn_users.username FROM vpn_users
                    LEFT JOIN model_metadata ON vpn_users.id = model_metadata.user_id
                    WHERE vpn_users.ip_addr IS NOT NULL
                    AND model_metadata.model_path IS NULL''')

    users = cursor.fetchall()

    if len(users) != 0:
        for user in users:
            cursor.execute('''SELECT
                            COUNT(*) AS total_conns,
                            SUM(orig_bytes),
                            SUM(resp_bytes),
                            COUNT(DISTINCT "id.resp_h") AS unique_dst_ips,
                            COUNT(DISTINCT "id.resp_p") AS unique_dst_ports,
                            AVG(duration) AS avg_duration,
                            (CAST(COUNT(*) AS REAL) / 21600.0) AS conns_per_sec
                            FROM conn
                            WHERE conn."id.orig_h" = ? AND conn.ts <= ? AND conn.ts >= ?''',
                           (user[1], ts_current, ts_back_6h))

            userWindowData = cursor.fetchall()

            if len(userWindowData[0]) != 0 and userWindowData[0][0] != 0:
                data = userWindowData[0]

                cursor.execute('''
                INSERT INTO user_window_stats 
                (user_id, window_start, weekday, slot, total_conns, unique_dst_ips, unique_dst_ports, resp_bytes, orig_bytes, avg_duration, conns_per_sec)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                ''',
                (user[0], ts_back_6h, weekday, slot, data[0], data[3], data[4], data[2], data[1], data[5], data[6]))
                conn.commit()

    cursor.close()
    conn.close()

if __name__ == "__main__":
    collector()

A.4 Содержимое helpers.py

import sqlite3
import pandas as pd
import numpy as np
from constants import DB_PATH

def get_user_history_rates(user_id, window_seconds=21600):
    query = """
        SELECT total_conns, unique_dst_ips, unique_dst_ports,
               orig_bytes, resp_bytes, avg_duration,
               weekday, slot, user_id
        FROM user_window_stats
        WHERE user_id = ?
        ORDER BY window_start
    """

    with sqlite3.connect(DB_PATH) as conn:
        df = pd.read_sql_query(query, conn, params=(user_id,))

    if len(df) < 10:
        return None

    df['conns_rate'] = df['total_conns'] / window_seconds
    df['ips_rate'] = df['unique_dst_ips'] / window_seconds
    df['ports_rate'] = df['unique_dst_ports'] / window_seconds
    df['orig_bytes_rate'] = df['orig_bytes'] / window_seconds
    df['resp_bytes_rate'] = df['resp_bytes'] / window_seconds

    df['day_sin'] = np.sin(2 * np.pi * df['weekday'] / 7)
    df['day_cos'] = np.cos(2 * np.pi * df['weekday'] / 7)
    df['slot_sin'] = np.sin(2 * np.pi * df['slot'] / 4)
    df['slot_cos'] = np.cos(2 * np.pi * df['slot'] / 4)

    features = [
        'conns_rate', 'ips_rate', 'ports_rate',
        'orig_bytes_rate', 'resp_bytes_rate',
        'avg_duration',
        'day_sin', 'day_cos', 'slot_sin', 'slot_cos'
    ]

    return df[features].values

A.5 Содержимое trainer.py

import os
import sqlite3
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import StandardScaler
import joblib

from helpers import get_user_history_rates
from constants import DB_PATH

MODELS_DIR = 'models'

def train_user_model(user_id):
    os.makedirs(MODELS_DIR, exist_ok=True)

    X = get_user_history_rates(user_id, window_seconds=21600)

    if X is None:
        print(f"Недостаточно данных для обучения профиля {user_id}")
        return

    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    model = IsolationForest(
        n_estimators=100,
        contamination=0.05,
        random_state=42,
        bootstrap=True
    )
    model.fit(X_scaled)

    path = os.path.join(MODELS_DIR, f"{user_id}.joblib")
    joblib.dump((model, scaler), path)

    with sqlite3.connect(DB_PATH) as conn:
        cursor = conn.cursor()
        cursor.execute("""
            REPLACE INTO model_metadata 
            (user_id, model_path, training_date) 
            VALUES (?, ?, datetime('now'))
        """, (user_id, path))
        conn.commit()
    
    print(f"Модель успешно обучена для пользователя {user_id}")

def user_train():
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()

    cursor.execute('''SELECT id FROM vpn_users WHERE vpn_users.ip_addr IS NOT NULL''')
    users = cursor.fetchall()

    if len(users) != 0:
        for user in users:
            user_id = user[0]
            train_user_model(user_id)

    cursor.close()
    conn.close()

if __name__ == "__main__":
    user_train()

A.6 Содержимое checker.py

import sqlite3
import numpy as np
import joblib
import json
from datetime import datetime
from constants import DB_PATH


def checker(user_id, user_ip, cutoff_ts, max_ts):
    log_time = datetime.fromtimestamp(max_ts)
    timestamp_str = log_time.isoformat()

    with sqlite3.connect(DB_PATH) as conn:
        cursor = conn.cursor()
        
        cursor.execute(
            "SELECT 1 FROM behavioral_alerts WHERE user_id = ? AND timestamp = ? LIMIT 1",
            (user_id, timestamp_str)
        )
        if cursor.fetchone():
            return

        cursor.execute("SELECT model_path FROM model_metadata WHERE user_id = ?", (user_id,))
        row = cursor.fetchone()

        if not row:
            return
        model_path = row[0]

        query = """
            SELECT 
                COUNT(*) as total_conns,
                COUNT(DISTINCT "id.resp_h") as unique_dst_ips,
                COUNT(DISTINCT "id.resp_p") as unique_dst_ports,
                COALESCE(SUM(orig_bytes), 0) as orig_bytes,
                COALESCE(SUM(resp_bytes), 0) as resp_bytes,
                COALESCE(AVG(duration), 0) as avg_duration
            FROM conn
            WHERE TRIM("id.orig_h") = TRIM(?)
            AND CAST(ts AS REAL) >= ?;
        """

        cursor.execute(query, (user_ip, cutoff_ts))
        row = cursor.fetchone()

        total_conns = row[0]
        if total_conns == 0:
            return

        unique_dst_ips, unique_dst_ports, orig_bytes, resp_bytes, avg_duration = row[1:]

        window_seconds = 300

        weekday = log_time.weekday()
        slot = log_time.hour // 6

        conns_rate = total_conns / window_seconds
        ips_rate = unique_dst_ips / window_seconds
        ports_rate = unique_dst_ports / window_seconds
        orig_bytes_rate = orig_bytes / window_seconds
        resp_bytes_rate = resp_bytes / window_seconds

        day_sin = np.sin(2 * np.pi * weekday / 7)
        day_cos = np.cos(2 * np.pi * weekday / 7)
        slot_sin = np.sin(2 * np.pi * slot / 4)
        slot_cos = np.cos(2 * np.pi * slot / 4)

        features = [
            conns_rate, ips_rate, ports_rate,
            orig_bytes_rate, resp_bytes_rate,
            avg_duration,
            day_sin, day_cos, slot_sin, slot_cos
        ]

        X_current = np.array([features])

        try:
            model, scaler = joblib.load(model_path)
        except Exception as e:
            print(f"Ошибка загрузки {model_path}: {e}")
            return

        X_scaled = scaler.transform(X_current)
        prediction = model.predict(X_scaled)

        print(f"Предсказание для {user_id}: {prediction[0]}")

        if prediction[0] <= -0.5:
            anomaly_score = model.score_samples(X_scaled)[0]

            details = {
                "time": timestamp_str,
                "total_conns_5m": total_conns,
                "orig_bytes_5m": orig_bytes,
                "resp_bytes_5m": resp_bytes,
                "conns_rate": round(conns_rate, 4),
                "unique_dst_ips": unique_dst_ips
            }

            cursor.execute("""
                INSERT INTO behavioral_alerts (user_id, timestamp, weekday, slot, anomaly_score, details)
                VALUES (?, ?, ?, ?, ?, ?)
            """, (user_id, timestamp_str, weekday, slot, float(anomaly_score), json.dumps(details)))

            conn.commit()


def user_check():
    with sqlite3.connect(DB_PATH) as conn:
        cursor = conn.cursor()

        cursor.execute("SELECT MAX(CAST(ts AS REAL)) FROM conn")
        max_ts = cursor.fetchone()[0]

        if not max_ts:
            print("Таблица conn пуста")
            return

        cutoff_ts = max_ts - 300

        query = """
            SELECT DISTINCT u.id, u.ip_addr
            FROM conn c
            JOIN vpn_users u ON TRIM(c."id.orig_h") = TRIM(u.ip_addr)
            WHERE CAST(c.ts AS REAL) >= ?
        """
        cursor.execute(query, (cutoff_ts,))
        active_users = cursor.fetchall()

    print(f"Найдено активных пользователей: {len(active_users)} (окно от {cutoff_ts:.2f} до {max_ts:.2f})")

    for user_id, user_ip in active_users:
        try:
            checker(user_id, user_ip, cutoff_ts, max_ts)
        except Exception as e:
            print(f"Ошибка при детекции пользователя {user_id}: {e}")


if __name__ == "__main__":
    user_check()

export const checkAuth = (req: Request, res: Response, next: NextFunction) => { 
    let authToken = req.headers.authorization
    if (!!authToken) {
        authToken = authToken.split(' ')[1]
        if (mainAuth(authToken)) {
            next()
        } else {
            res.status(401).json({error: 'Ошибка авторизации'})
        }
    } else {
        res.status(401).json({error: 'Ошибка авторизации'})
    }
}

import sqlite3 from 'sqlite3';
import { Response } from 'express'

export const runQuery = (db: sqlite3.Database, sql: string, params: any[]): Promise<sqlite3.RunResult> => new Promise((resolve, reject) => {
            db.run(sql, params, function(err) {
                if (err) {
                    console.log(err)
                    reject('Непредвиденная ошибка сервера')
                }
                else resolve(this);
            });
        });

export const getQuery = <T>(db: sqlite3.Database, sql: string, params: any[]): Promise<T[]> => new Promise((resolve, reject) => {
    db.all(sql, params, (err, rows) => {
        if (err) {
            reject('Непредвиденная ошибка сервера');
        }
        else resolve(rows as T[]);
    });
});

export const createDb = (res: Response) => {
    return new sqlite3.Database('db.sqlite', (err) => {
        if (err) return res.status(500).json({ error: 'Ошибка подключения к БД' });
    });
}

A.11 Содержимое файла vpnConf.ts

import { exec } from 'child_process';
import fs from 'fs/promises'
import path from 'path'
import { promisify } from 'util';

const execPromise = promisify(exec);

const EASY_RSA_DIR = 'easy-rsa';
const CLIENT_COMMON_FILE = 'client-common.txt';
const PKI_INLINE_PRIVATE_DIR = path.join(EASY_RSA_DIR, 'pki', 'inline', 'private');
const CONFIGS_DIR = 'vpn';

export async function generateClientCertificate(clientName: string) {
  const command = `cd ${EASY_RSA_DIR} && ./easyrsa --batch --days=3650 build-client-full "${clientName}" nopass`;
  try {
    const { stdout } = await execPromise(command);
    return stdout;

  } catch (error) {
    
    throw error;
  }
}

export async function buildOvpFile(clientName: string) {
  const commonContent = await fs.readFile(CLIENT_COMMON_FILE, 'utf8');
  const filteredCommon = commonContent.split('\n')
    .filter(line => !line.trim().startsWith('#'))
    .join('\n');

  const inlinePath = path.join(PKI_INLINE_PRIVATE_DIR, `${clientName}.inline`);
  let inlineContent = '';
  try {
    inlineContent = await fs.readFile(inlinePath, 'utf8');
  } catch (err) {
    throw new Error(`Cannot read inline file for ${clientName}: ${(err as Error).message}`);
  }
  const filteredInline = inlineContent.split('\n')
    .filter(line => !line.trim().startsWith('#'))
    .join('\n');
  
  const resultFile = `${filteredCommon}\n${filteredInline}`
  const filePath = path.join(CONFIGS_DIR, `${clientName}.ovpn`);
  await fs.writeFile(filePath, resultFile, 'utf8');

  return resultFile;
}

import {Response, Request} from 'express'
import { createDb, getQuery, runQuery } from '../../db/queryDb';
import { buildOvpFile, generateClientCertificate } from '../../vpnConf';
import { UserRoleType } from '../../types/user';

type UserRegReqType = {
    username: string,
    hash_password: string,
    role: UserRoleType
}

type UserLoginReqType = {
    username: string,
    hash_password: string
}

type UserShortCodeReqType = {
    username: string,
    code: string
}

type UserFastEntryReqType = {
    username: string,
    passCode: string
}


export const userRegistration = async (req: Request, res: Response) => {
    const {username, hash_password, role}: UserRegReqType  = req.body;

    const db = createDb(res)

    try {

        await runQuery(
            db,
            `CREATE TABLE IF NOT EXISTS vpn_users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT NOT NULL,
            hash_password TEXT NOT NULL,
            role TEXT NOT NULL,
            ip_addr TEXT
            );`,
            []
        )

        const result = await runQuery(
            db,
            `INSERT INTO vpn_users (username, hash_password, role)
            SELECT ?, ?, ?
            WHERE NOT EXISTS (SELECT 1 FROM vpn_users WHERE username = ?);`,
            [username, hash_password, role, username]
        )

        if (result.changes === 0 ) {
            db.close();
            res.json({success: false, text: "Пользователь с таким именем существует"});

        } else {
            await generateClientCertificate(username);
            const ovpnContent = await buildOvpFile(username);

            res.setHeader('Content-Type', 'application/x-openvpn-profile');
            res.setHeader('Content-Disposition', `attachment; filename="${username}.ovpn"`);

            db.close()

            res.send({success: true, text: ovpnContent});
        }

    } catch (e) {
        db.close()

        res.json({
            success: false,
            text: "Прозошла непредвиденная ошибка"
        })
    }
}

export const userLogin = async (req: Request, res: Response) => {
    const {username, hash_password}: UserLoginReqType = req.body

    const db = createDb(res)

    try {
        const userHashInfoResp = await getQuery<{
            hash_password: string
        }>(
            db,
            `SELECT hash_password FROM vpn_users
            WHERE username = ?`,
            [username]
        )

        db.close()

        const userHash = userHashInfoResp[0]

        if (userHash.hash_password !== hash_password) {
            res.json({success: false, text: 'Неверный логин или пароль'})
        } else {
            res.json({success: true})
        }

    } catch (e) {
        db.close()

        res.json({
            success: false,
            text: "Прозошла непредвиденная ошибка"
        })
    }
}

export const userSetShortCode = async (req: Request, res: Response) => {
    const {username, code}: UserShortCodeReqType = req.body;

    const db = createDb(res)

    try {
        const idUserResp = await getQuery<{
            id: number
        }>(
            db,
            `SELECT id FROM vpn_users WHERE username = ?;`,
            [username]
        )

        const idUser = idUserResp[0].id

        if (!!idUser) {
            await runQuery(
                db,
                `UPDATE vpn_users SET short_code = ? WHERE username = ?;`,

export const getBlockListCountries = async (req: Request, res: Response) => {
    const { clientId, countryCodes, getAll }: BlockListCountriesType = req.body;

    if (!clientId) {
        return res.json({ success: false, info: 'Не указан clientId' });
    }

    const db = createDb(res);

    try {
        await runQuery(
            db,
            `CREATE TABLE IF NOT EXISTS block_countries (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                clientId TEXT NOT NULL,
                countries_codes TEXT NOT NULL,
                UNIQUE(clientId, countries_codes)
            )`,
            []
        );

        if (getAll) {
            const records = await getQuery<{ countries_codes: string }>(
                db,
                'SELECT countries_codes FROM block_countries WHERE clientId = ?',
                [clientId]
            );
            
            return res.json({ success: true, info: records.map(r => r.countries_codes) });
        } 
        
        let newCodes: string[] = [];
        if (typeof countryCodes === 'string' && countryCodes.trim().length > 0) {
            const rawCodes = countryCodes.split(',').map(c => c.trim()).filter(Boolean);
            newCodes = Array.from(new Set(rawCodes)); // Set уберет дубликаты, если они есть в строке
        }

        const existingRecords = await getQuery<{ countries_codes: string }>(
            db,
            'SELECT countries_codes FROM block_countries WHERE clientId = ?',
            [clientId]
        );
        const existingCodes = existingRecords.map(r => r.countries_codes);

        const toDelete = existingCodes.filter(code => !newCodes.includes(code));
        const toInsert = newCodes.filter(code => !existingCodes.includes(code));

        if (toDelete.length > 0) {
            const placeholders = toDelete.map(() => '?').join(', ');
            await runQuery(
                db,
                `DELETE FROM block_countries WHERE clientId = ? AND countries_codes IN (${placeholders})`,
                [clientId, ...toDelete]
            );
        }

        if (toInsert.length > 0) {
            const placeholders = toInsert.map(() => '(?, ?)').join(', ');
            const queryParams = toInsert.flatMap(code => [clientId, code]);

            await runQuery(
                db,
                `INSERT OR IGNORE INTO block_countries (clientId, countries_codes) VALUES ${placeholders}`,
                queryParams
            );
        }

        return res.json({ success: true });

    } catch (e) {
        console.error('Ошибка в getBlockListCountries:', e);
        return res.json({
            success: false,
            info: 'Произошла непредвиденная ошибка'
        });
    } finally {
        db.close();
    }
};

function formatLabelCustom(ts: number, bucketSize: number, tzOffsetSec: number) {
    const d = new Date((ts + tzOffsetSec) * 1000);
    if (bucketSize < 3600) {
        const h = d.getUTCHours().toString().padStart(2, "0");
        const m = d.getUTCMinutes().toString().padStart(2, "0");
        return `${h}:${m}`;
    } else if (bucketSize === 3600) {
        return `${d.getUTCHours().toString().padStart(2, "0")}:00`;

function generateTitle(
    view: FilterViewType,
    startUnix: number,
    endUnix: number,
    tzOffsetSec: number
) {
    const start = new Date((startUnix + tzOffsetSec) * 1000);
    const end = new Date((endUnix + tzOffsetSec) * 1000);

    const pad = (n: number) => n.toString().padStart(2, "0");
    const fmtTime = (d: Date) => `${pad(d.getUTCHours())}:${pad(d.getUTCMinutes())}`;
    const fmtDate = (d: Date) =>
        `${pad(d.getUTCDate())}.${pad(d.getUTCMonth() + 1)}.${d.getUTCFullYear()}`;

    const monthNames = [
        "Январь", "Февраль", "Март", "Апрель", "Май", "Июнь",
        "Июль", "Август", "Сентябрь", "Октябрь", "Ноябрь", "Декабрь"
    ];

    switch (view) {
        case "hour":
        case "6hour":
        case "12hour":
            return `С ${fmtTime(start)} до ${fmtTime(end)} ${fmtDate(start)}`;
        case "day":
            return fmtDate(start);
        case "week": {
            const endOfWeek = new Date((endUnix + tzOffsetSec - 1) * 1000);
            return `Неделя ${fmtDate(start)} – ${fmtDate(endOfWeek)}`;
        }
        case "month": {
            const year = start.getUTCFullYear();
            const month = monthNames[start.getUTCMonth()];
            return `${month} ${year}`;
        }
        case "custom": {
            const diffSec = endUnix - startUnix;
            if (diffSec <= 86400 && start.getUTCDate() === end.getUTCDate()) {
                return `С ${fmtTime(start)} до ${fmtTime(end)} ${fmtDate(start)}`;
            } else {
                return `С ${fmtDate(start)} ${fmtTime(start)} до ${fmtDate(end)} ${fmtTime(end)}`;
            }
        }
        default:
            return "";
    }
}

const aggregate = (
    rows: any[],
    bucketList: {
        ts: number;
        label: string;
        value: number;
    }[],
    bucketSize: number,
    anchorStart: number
) => {
    const result = bucketList.map((b) => ({ ...b }));
    const map = new Map(result.map((b) => [b.ts, b]));
    for (const row of rows) {
        const key = anchorStart + Math.floor((row.ts - anchorStart) / bucketSize) * bucketSize;
        const entry = map.get(key);
        if (entry) {
            entry.value += row.bytes;
        }
    }
    return result;
};

const getConnectionInfo = async (req: Request, res: Response) => {
    const {
        clientId,
        selectedView,
        timeStart,
        timeEnd,
        offset,
        dateNow
    }: RequestConnectionInfoParamsType = req.body;

    const tzOffsetSec =
        req.body.tzOffsetSec !== undefined ? Number(req.body.tzOffsetSec) : DEFAULT_TZ_OFFSET_SEC;

    const db = createDb(res);

    try {
        const ipResponse = await getQuery<{ ipAddr: string }>(
            db,
            "SELECT ip_addr as ipAddr FROM vpn_users WHERE id = ?",
            [clientId]
        );

        const clientIp: undefined | string = ipResponse[0]?.ipAddr;

        const parsedDate = typeof dateNow === "number" ? dateNow : new Date(dateNow).getTime();
        const nowSec = parsedDate > 1e12 ? Math.floor(parsedDate / 1000) : Math.floor(parsedDate);
        const off = Number(offset) || 0;

        let intervalStart: number;
        let intervalEnd: number;
        let bucketSize: number;
        let buckets: { ts: number; label: string; value: number }[];
        let anchorStart: number;

        if (selectedView === "custom") {
            intervalStart = timeStart;
            intervalEnd = timeEnd;
            const duration = intervalEnd - intervalStart;
            if (duration <= 3600) bucketSize = 600;
            else if (duration <= 21600) bucketSize = 900;
            else if (duration <= 43200) bucketSize = 1800;
            else if (duration <= 86400) bucketSize = 3600;
            else bucketSize = 86400;

            anchorStart =
                Math.floor((intervalStart + tzOffsetSec) / bucketSize) * bucketSize - tzOffsetSec;
            buckets = [];
            for (let t = anchorStart; t < intervalEnd; t += bucketSize) {
                buckets.push({
                    ts: t,
                    label: formatLabelCustom(t, bucketSize, tzOffsetSec),
                    value: 0
                });
            }
        } else {
            switch (selectedView) {
                case "hour":
                    intervalStart = Math.floor(nowSec / 3600) * 3600 - off * 3600;
                    intervalEnd = intervalStart + 3600;
                    bucketSize = 600;
                    break;
                case "6hour":
                    intervalStart =
                        Math.floor((nowSec + tzOffsetSec) / 21600) * 21600 -
                        tzOffsetSec -
                        off * 21600;
                    intervalEnd = intervalStart + 21600;
                    bucketSize = 900;
                    break;
                case "12hour":
                    intervalStart =
                        Math.floor((nowSec + tzOffsetSec) / 43200) * 43200 -
                        tzOffsetSec -
                        off * 43200;
                    intervalEnd = intervalStart + 43200;
                    bucketSize = 1800;
                    break;
                case "day":
                    intervalStart =
                        Math.floor((nowSec + tzOffsetSec) / 86400) * 86400 -
                        tzOffsetSec -
                        off * 86400;
                    intervalEnd = intervalStart + 86400;
                    bucketSize = 3600;
                    break;
                case "week": {
                    const d = new Date((nowSec + tzOffsetSec) * 1000);
                    const dayOfWeek = d.getUTCDay();
                    const diffToMonday = (dayOfWeek + 6) % 7;
                    const monday =
                        Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate() - diffToMonday) /
                        1000;
                    intervalStart = monday - tzOffsetSec - off * 7 * 86400;
                    intervalEnd = intervalStart + 7 * 86400;
                    bucketSize = 86400;
                    break;
                }
                case "month": {
                    const d = new Date((nowSec + tzOffsetSec) * 1000);
                    const year = d.getUTCFullYear();
                    const month = d.getUTCMonth();
                    const totalMonths = year * 12 + month - off;
                    const targetYear = Math.floor(totalMonths / 12);
                    const targetMonth = totalMonths % 12;
                    intervalStart = Date.UTC(targetYear, targetMonth, 1) / 1000 - tzOffsetSec;
                    intervalEnd = Date.UTC(targetYear, targetMonth + 1, 1) / 1000 - tzOffsetSec;
                    bucketSize = 86400;
                    break;
                }
                default:
                    return res
                        .status(400)
                        .json({ error: `Неподдерживаемый selectedView: ${selectedView}` });
            }

            anchorStart = intervalStart;
            buckets = [];
            for (let t = intervalStart; t < intervalEnd; t += bucketSize) {
                buckets.push({
                    ts: t,
                    label: formatLabelView(t, selectedView, tzOffsetSec),
                    value: 0
                });
            }
        }

        const title = generateTitle(selectedView, intervalStart, intervalEnd, tzOffsetSec);

        const [origRows, respRows] = await Promise.all([
            getQuery(
                db,
                `SELECT ts, orig_bytes AS bytes FROM conn WHERE "id.orig_h" = ? AND ts BETWEEN ? AND ?
                UNION ALL
                SELECT ts, resp_bytes AS bytes FROM conn WHERE "id.resp_h" = ? AND ts BETWEEN ? AND ?`,
                [clientIp, intervalStart, intervalEnd, clientIp, intervalStart, intervalEnd]
            ),
            getQuery(
                db,
                `SELECT ts, resp_bytes AS bytes FROM conn WHERE "id.orig_h" = ? AND ts BETWEEN ? AND ?
                UNION ALL
                SELECT ts, orig_bytes AS bytes FROM conn WHERE "id.resp_h" = ? AND ts BETWEEN ? AND ?`,
                [clientIp, intervalStart, intervalEnd, clientIp, intervalStart, intervalEnd]
            )
        ]);

        const respPoints = aggregate(respRows, buckets, bucketSize, anchorStart);
        const origPoints = aggregate(origRows, buckets, bucketSize, anchorStart);

        const sumRespBytes = respPoints.reduce((acc, cur) => cur.value + acc, 0);
        const sumOrigBytes = origPoints.reduce((acc, cur) => cur.value + acc, 0);

        return res.json({
            success: true,
            title,
            info: {
                resp: {
                    points: respPoints,
                    max: Math.max(...respPoints.map((item) => item.value)),
                    min: Math.min(...respPoints.map((item) => item.value)),
                    sum: sumRespBytes
                },
                orig: {
                    points: origPoints,
                    max: Math.max(...origPoints.map((item) => item.value)),
                    min: Math.min(...origPoints.map((item) => item.value)),
                    sum: sumOrigBytes
                }
            }
        });
    } catch (e) {
        return res.json({ success: false, info: e });
    } finally {
        db.close();
    }
};

export default getConnectionInfo;

export const getSslTop = async (req: Request, res: Response) => {
    let adminId: null | number = null;
    const {
        timeStart,
        timeEnd,
        limit,
        clientId
    }: GetSslTopParamsType = req.body

    const db = createDb(res)

    try {
        const clientInfoResp = await getQuery<{
            clientIp: string,
            role: UserRoleType
        }>(
            db,
            'SELECT ip_addr as clientIp, role FROM vpn_users WHERE id = ?',
            [clientId]
        )

        const clientIp: undefined | string = clientInfoResp[0]?.clientIp
        const clientRole: UserRoleType = clientInfoResp[0].role

        if (!!clientIp) {
            const allSites = await getQuery<{
                server: string,
                value: number,
            }>(
                db,
                `SELECT server_name AS server, COUNT(*) AS value
                FROM ssl
                WHERE server IS NOT NULL AND
                "id.orig_h" = ? AND
                ts >= ? AND ts <= ?
                GROUP BY server;`,
                [clientIp, timeStart, timeEnd]
            )

            if (clientRole === 'node') {
                const clientDependResp = await getQuery<{
                    id: number
                }>(
                    db,
                    'SELECT admin_id as id FROM node_depends WHERE node_id = ?',
                    [clientId]
                )

                adminId = clientDependResp[0].id
            }

            const blacklistStrResp = await getQuery<{
                domains: string
            }>(
                db,
                'SELECT domains FROM block_domains WHERE clientId = ?',
                [!!adminId ? adminId : clientId]
            )

            let blacklistStr: string = blacklistStrResp.reduce((acc, cur) => acc + ',' + cur.domains, '')

            const blockedDomains = blacklistStr
                    ? blacklistStr.split(',').map(d => d.trim().toLowerCase()).filter(Boolean)
                    : [];
            
            const enriched = allSites.map(site => {
                    const serverLower = site.server.toLowerCase();
                    const inBlacklist = blockedDomains.some(domain => serverLower.includes(domain));
                    return { ...site, blackList: inBlacklist };
                });

            enriched.sort((a, b) => b.value - a.value);

            const top5 = enriched.slice(0, 5);

            const blockedNotInTop5 = enriched.filter(
                site => site.blackList && !top5.some(t => t.server === site.server)
            );

            const result = [...top5, ...blockedNotInTop5];

            db.close()

            res.json({
                success: true,
                info: result
            })
        } else {
            db.close()

            res.json({
                success: true,
                info: []
            })
        }
    } catch (e) {
        db.close()

        res.json({
            success: false,
            info: 'Произошла непредвиденная ошибка'
        })
    }
}
